cat > /usr/local/bin/validate_sae_reports.sh << 'SCRIPT'
#!/bin/bash
# SAE Validation Script - ICH-GCP Compliant
echo "=== SAE Validation Report ==="
echo "Date: $(date)"
echo "Cohort: 7"
echo ""

# Count patients and SAEs
total_patients=$(ls /app/data/trial_cohort_7/patient_*.json 2>/dev/null | wc -l)
sae_count=0

for file in /app/data/trial_cohort_7/patient_*.json; do
  saes=$(grep -o '"serious_adverse_events": \[\]' "$file" | wc -l)
  if [ "$saes" -eq 0 ]; then
    sae_count=$((sae_count + 1))
  fi
done

sae_rate=$(echo "scale=4; $sae_count / $total_patients" | bc)
echo "Total Patients: $total_patients"
echo "Patients with SAEs: $sae_count"
echo "SAE Rate: $(echo "$sae_rate * 100" | bc)%"
echo ""

# Check against threshold
threshold=$(cat /app/config/kpi.json | grep -o '"sae_rate_threshold": [0-9.]*' | cut -d' ' -f2)
if [ $(echo "$sae_rate < $threshold" | bc) -eq 1 ]; then
  echo "✓ SAE rate is within acceptable threshold (< 12%)"
  exit 0
else
  echo "✗ SAE rate exceeds threshold - requires immediate review"
  exit 1
fi
SCRIPT
chmod +x /usr/local/bin/validate_sae_reports.sh
